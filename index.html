async function chat() { const input = document.getElementById("chatInput"); const output = document.getElementById("chatOutput"); const message = input.value.trim(); if (!message) { output.textContent = "براہ کرم اپنا سوال لکھیں۔"; return; } output.textContent = "🤖 NEXORA AI جواب تیار کر رہا ہے..."; input.disabled = true; try { const response = await fetch( "https://abdul-ai-server.abdulsattar76556788.workers.dev/", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ message: message }) } ); const data = await response.json(); if (!response.ok) { throw new Error(data.error || "API connection failed"); } output.textContent = data.answer || "AI سے جواب موصول نہیں ہوا۔"; } catch (error) { output.textContent = "معذرت! AI کنکشن میں مسئلہ ہے۔ دوبارہ کوشش کریں۔\n" + error.message; } finally { input.disabled = false; } }
